#!/bin/bash

################################################################################
# Script Name: modify_sql_file.sh
# Description: This script modifies an SQL file by replacing specific lines.
#
# Requirements:
#   - The script expects two command-line arguments:
#       1. Input SQL file path
#       2. New tablespace name
#   - The input SQL file should have the following format:
#       - Lines containing 'CREATE TABLE "FCUBSLIVE"' will be modified to 'ALTER TABLE "ABFCUBSLIVE" MODIFY'.
#       - Lines containing 'PARTITION BY LIST' will be appended as-is.
#       - Lines containing 'PARTITION' will have the content after the first ')' truncated.
#       - Lines containing 'TABLESPACE' will be replaced with the new tablespace name.
#         - Two types of TABLESPACE lines are supported:
#             1. TABLESPACE "old_value",
#             2. TABLESPACE "old_value" )  ENABLE ROW MOVEMENT ;
#
# Usage: ./modify_sql_file.sh <input.sql> <tablespace>
#   - <input.sql>: Path to the input SQL file.
#   - <tablespace>: New tablespace name.
################################################################################

# Check if the correct number of arguments is provided
if [ "$#" -ne 2 ]; then
    echo "Usage: $0 <input.sql> <tablespace>"
    exit 1
fi

# Input SQL file
input_file="$1"
tablespace="$2"

# Output file name with ZB instead of TB
output_file="${input_file//TB/ZB}"

# Grep lines, modify content, then write to output file
grep 'CREATE TABLE "FCUBSLIVE"' "$input_file" | sed 's/CREATE/ALTER/' | sed 's/FCUBSLIVE/ABFCUBSLIVE/' | sed "s/TB/ZB/g" | sed 's/$/ MODIFY/' > "$output_file"

grep 'PARTITION BY LIST' "$input_file" | head -n 1 >> "$output_file"

while IFS= read -r line; do
    if echo "$line" | grep -q 'PARTITION'; then
        echo "$line" | sed 's/).*/)/' >> "$output_file"
    fi
    if echo "$line" | grep -q 'TABLESPACE'; then
        if echo "$line" | grep -q 'ENABLE ROW MOVEMENT'; then
            echo "TABLESPACE \"$tablespace\" )  ENABLE ROW MOVEMENT ;" >> "$output_file"
        else
            echo "$line" | sed "s/TABLESPACE.*,/TABLESPACE \"$tablespace\",/" >> "$output_file"
        fi
    fi
done < <(grep -v 'PARTITION BY LIST' "$input_file")

# Remove 2nd and 3rd lines from the output file
sed -i '2,3d' "$output_file"

#!/bin/bash

# Function to display usage information
usage() {
    echo "Usage: $0 [duration_in_minutes]"
    exit 1
}

# Check if duration is provided, otherwise set default to 15 minutes
if [ $# -eq 0 ]; then
    duration=15
elif [ $# -eq 1 ]; then
    duration=$1
else
    usage
fi

# Get current time and calculate start time
end_time=$(date +%s)
start_time=$((end_time - (duration * 60)))

# Function to display system stats
display_stats() {
    echo "===== $1 ====="
    echo "Total Available: $2"
    echo "Current Utilization: $3"
    echo "Peak in the last $duration minutes: $4"
    echo "Top 10 Processes:"
    echo "PID   CPU%   MEM%   COMMAND"
    ps aux --sort=-%cpu | head -11 | awk '{print $2, $3, $4, $11}'
    echo "Process Details:"
    ps aux --sort=-%cpu | head -11
    echo
}

# CPU Statistics
display_stats "CPU" \
    "$(sar -u -s $(date -d @$start_time +"%H:%M:%S") -e $(date -d @$end_time +"%H:%M:%S") | grep 'Average.*all' | awk '{print $NF}')" \
    "$(mpstat -P ALL 1 $duration | tail -n 1 | awk '{print (100 - $NF) "%"}')" \
    "$(sar -u -s $(date -d @$start_time +"%H:%M:%S") -e $(date -d @$end_time +"%H:%M:%S") | grep 'Average.*all' | awk '{print $NF}')"

# Memory Statistics
display_stats "Memory" \
    "$(sar -r -s $(date -d @$start_time +"%H:%M:%S") -e $(date -d @$end_time +"%H:%M:%S") | grep 'Average' | awk '{print $2 " kB"}')" \
    "$(free -m | grep Mem | awk '{print $3 " MB"}')" \
    "$(free -m | grep Mem | awk '{print $7 " MB"}')"

# IO Statistics

    "$(sar -b -s $(date -d @$start_time +"%H:%M:%S") -e $(date -d @$end_time +"%H:%M:%S") | grep 'Average' | awk '{print $6 " kB"}')" \

    "$(sar -n DEV -s $(date -d @$start_time +"%H:%M:%S") -e $(date -d @$end_time +"%H:%M:%S") | grep 'Average.*lo' | awk '{print $NF " kB/s"}')" \

#!/bin/bash

# Ensure the file with the list of files to delete is provided as a command-line argument
if [ "$#" -ne 1 ]; then
    echo "Usage: $0 <file_list>"
    exit 1
fi

file_list="$1"

# Check if the file exists
if [ ! -e "$file_list" ]; then
    echo "File $file_list not found."
    exit 1
fi

# Read each line from the file and delete the corresponding file
while IFS= read -r file_path; do
    # Check if the file exists before attempting to delete
    if [ -e "$file_path" ]; then
        rm -f "$file_path"
        echo "Deleted: $file_path"
    else
        echo "File not found: $file_path"
    fi
done < "$file_list"
